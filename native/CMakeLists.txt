cmake_minimum_required(VERSION 3.20)
project(TorrentNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS ON)
if(NOT DEFINED TTD_LIBTORRENT_SOURCE)
  message(FATAL_ERROR "Set TTD_LIBTORRENT_SOURCE to the pinned libtorrent source")
endif()
add_subdirectory("${TTD_LIBTORRENT_SOURCE}" libtorrent EXCLUDE_FROM_ALL)
add_library(TorrentNative STATIC src/TorrentNative.cpp)
target_include_directories(TorrentNative PUBLIC include)
target_link_libraries(TorrentNative PUBLIC torrent-rasterbar)
option(TTD_BUILD_TESTS "Run real cached-payload bridge tests" OFF)
if(TTD_BUILD_TESTS)
  enable_testing()
  add_executable(TorrentNativeTests tests/TorrentNativeTests.cpp)
  target_link_libraries(TorrentNativeTests PRIVATE TorrentNative)
  target_compile_definitions(TorrentNativeTests PRIVATE TTD_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME TorrentNativeTests COMMAND TorrentNativeTests)
  set_tests_properties(TorrentNativeTests PROPERTIES TIMEOUT 40)
endif()
